# Python dependencies for the combined NIDS JupyterHub image.
#
# The image is the only environment the hub modules have, so this file must cover every
# module whose venue includes "nrp" -- not just the four it originally listed. ASN and BGP
# are also here: they run on the hub as well as on a laptop. The per-module lists live in
# assignments/registry.toml under [<CODE>.environment]; that file is the source of truth
# and this one is its union. Adding a module, or changing its `extra`, means editing here
# too.
#
#   - nids-asn-introduction  : pandas, matplotlib, requests  (also venue "local")
#   - nids-bgp-control-plane : pybgpkit-parser, pelicanfs, py-radix, pandas, matplotlib
#                              (also venue "local")
#   - nids-irr-rpki-whois    : py-radix, tqdm
#   - nids-itdk              : sqlalchemy, psycopg2-binary, pycountry, scipy, python-dotenv
#   - nids-telescope-traffic : dpkt, pandas, pyarrow, pybgpkit-parser, pelicanfs,
#                              py-radix, geoip2, maxminddb, matplotlib
#   - nids-dns-ecosystem     : pyspark, tldextract, dnspython, numpy, requests, matplotlib
#   - nids-iyp               : neo4j, python-dotenv, pandas, matplotlib
#
# nids-ucsdnt-expanse is deliberately absent: its venue is "expanse" (Slurm on SDSC),
# not this image.
#
# Version specifiers below come from nids-telescope-traffic/.../requirements.txt
# (the only assignment that pins). The rest are left unpinned to match the notebooks'
# bare `%pip install`; pin them once versions are confirmed against a real run.

# --- Telescope + BGP (pinned by the telescope assignment) ---
dpkt>=1.9
pandas>=2.0
pyarrow>=14.0
matplotlib>=3.5
geoip2>=4.7
maxminddb>=2.5
# py-radix, NOT pytricia: pytricia is source-only, so it needs a compiler on every
# platform and has no Windows wheel at all. py-radix ships cp39-cp314 wheels for
# win32/win_amd64, macOS and manylinux/musllinux. Imports as `radix`.
py-radix>=1.1
pelicanfs>=1.0
pybgpkit-parser>=0.18   # import as `import pybgpkit_parser as bgpkit` (NOT `import bgpkit`)

# --- DNS ecosystem ---
# OPEN ITEM: the all-spark-notebook base already ships Spark, and the PyPI `pyspark`
# distribution is source-only and large, so pip builds and installs a second copy over
# it. That costs build time and image size, and risks a version skew against $SPARK_HOME
# and the pre-staged hadoop-aws JAR. Likely wants removing or pinning to the base's
# Spark version — needs a real Spark session against the DNS notebook to confirm first.
pyspark
tldextract
dnspython               # import as `import dns.resolver`
numpy
requests

# --- Registries: WHOIS, IRR & RPKI ---
# py-radix is already pinned above. tqdm is the only addition; both are declared in
# registry.toml [IRR.environment].
tqdm

# --- ITDK ---
# From registry.toml [ITDK.environment]. The module is release = "later" and
# status = "blocked" on an in-namespace Postgres, but its venue is "nrp" only, so the
# image is where these have to be. python-dotenv is shared with IYP, below.
sqlalchemy
psycopg2-binary         # binary wheel: no libpq/compiler needed at image build time
pycountry
scipy

# --- IYP (Internet Yellow Pages) ---
# pandas + matplotlib are already covered above; jupyterlab comes from the base image.
# NOTE: the notebook also needs a reachable Neo4j instance and a neo4j_credentials.env
# file — neither is provided by this image, and neither is deployed anywhere yet.
neo4j                   # Neo4j Python driver, connects over bolt://
python-dotenv           # loads neo4j_credentials.env
