# Python dependencies for the combined NIDS JupyterHub image.

# --- Telescope + BGP (pinned by the telescope assignment) ---
dpkt>=1.9
pandas>=2.0
pyarrow>=14.0
matplotlib>=3.5

maxminddb>=2.5
# py-radix, NOT pytricia: pytricia is source-only, so it needs a compiler on every
# platform and has no Windows wheel at all. py-radix ships cp39-cp314 wheels for
# win32/win_amd64, macOS and manylinux/musllinux. Imports as `radix`.
py-radix>=1.1
pelicanfs>=1.0
pybgpkit-parser>=0.18   # import as `import pybgpkit_parser as bgpkit` (NOT `import bgpkit`)

pyspark
tldextract
dnspython               # import as `import dns.resolver`
numpy
requests
